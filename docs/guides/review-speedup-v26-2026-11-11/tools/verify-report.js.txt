'use strict';
// V2-6: independent spot-check of the report's numbers. It does NOT use the analyzers (v26-analyze.js, v26-report.js): every figure below is recomputed
// from the raw run files (trace.jsonl, summary.json, dispatch.log.jsonl, load-samples.jsonl, the stored walltime rows) and then looked up in the report
// text (in the row of the run, for per-run figures). Prints one line per check and exits 1 on any mismatch. Nothing here opens a session or transcript.
const fs = require('fs');
const path = require('path');
const HOME = process.env.HOME;
const RUNS = path.join(HOME, '.omp/pr-review-speedup/v26/runs');
const SESSION = fs.readFileSync(path.join(HOME, '.omp/pr-review-speedup/v26/load/load.log'), 'utf8').split('\n').map((l) => /^(\S+Z) load1=([\d.]+)/.exec(l)).filter(Boolean).map((m) => ({ t: Date.parse(m[1]), load1: Number(m[2]) }));
const REPORT = fs.readFileSync('/Users/ldangelo/Development/Sunstone/ensemble.prs-v26/docs/guides/review-speedup-v26-2026-11-11.md', 'utf8');
const EVID = '/Users/ldangelo/Development/Sunstone/ensemble.prs-v26/docs/guides/review-speedup-v26-2026-11-11';
const lines = REPORT.split('\n');
const rd = (f) => JSON.parse(fs.readFileSync(f, 'utf8'));
const f2 = (x) => (Math.round(x * 100) / 100).toFixed(2);
const f1 = (x) => (Math.round(x * 10) / 10).toFixed(1);
const IDS = ['v26-01-pr207-old-first', 'v26-02-pr207-new-first', 'v26-03-pr207-new-first-rep', 'v26-04-pr319-new-first', 'v26-05-pr319-old-first', 'v26-06-pr314-old-first', 'v26-08-pr314-new-first-redo4', 'v26-09-pr315-new-first', 'v26-10-pr315-old-first'];

function readTrace(dir) {
  const spans = new Map();
  for (const l of fs.readFileSync(path.join(dir, 'trace.jsonl'), 'utf8').split('\n')) {
    if (!l.trim()) continue;
    const e = JSON.parse(l);
    if (e.type === 'span_start') spans.set(e.span_id, { name: e.name, start: Date.parse(e.start_ts), attrs: e.attrs || {} });
    else if (e.type === 'span_end') { const s = spans.get(e.span_id); if (s) { s.end = Date.parse(e.end_ts); s.status = e.status; const lm = e.attrs && e.attrs.last_message_ts; s.iend = lm ? Date.parse(lm) : s.end; } }
  }
  return [...spans.values()];
}
function rowOf(id) { return lines.find((l) => l.startsWith('|') && l.includes(`\`${id}\``) && /\|\s*\d+\s*\|/.test(l)); }
function rowsOf(id) { return lines.filter((l) => l.startsWith('|') && l.includes(`\`${id}\``)); }
let bad = 0; const out = [];
function check(label, value, where) {
  const text = where === undefined ? REPORT : where;
  const ok = text.includes(String(value));
  if (!ok) bad += 1;
  out.push(`${ok ? 'OK  ' : 'FAIL'} ${label}: ${value}`);
}

const raw = {};
for (const id of IDS) {
  const dir = path.join(RUNS, id);
  const spans = readTrace(dir);
  const root = spans.find((s) => s.name === 'review' && !s.attrs.parent);
  const ends = spans.filter((s) => s.iend).map((s) => s.iend); const starts = spans.map((s) => s.start);
  const roots = spans.filter((s) => s.name === 'review'); if (roots.length !== 1) throw new Error('expected one root review span in ' + id); const wallMin = (roots[0].end - roots[0].start) / 60000;
  const ver = spans.filter((s) => s.name === 'review.verifier');
  const verStart = Math.min(...ver.map((s) => s.start)); const verEnd = Math.max(...ver.map((s) => s.iend));
  const stopped = ver.filter((s) => s.status === 'expired');
  const rev = spans.filter((s) => s.name === 'review.reviewer' || s.name === 'review.security-lane');
  const reviewRoundMin = (Math.max(...rev.map((s) => s.iend)) - Math.min(...rev.map((s) => s.start))) / 60000;
  const summary = rd(path.join(dir, 'summary.json'));
  const cands = summary.ledger.candidates;
  const unverified = cands.filter((c) => c.outcome === 'unverified').length;
  const dispatch = fs.readFileSync(path.join(dir, 'dispatch.log.jsonl'), 'utf8').split('\n').filter(Boolean).map((l) => JSON.parse(l));
  const verAgents = dispatch.flatMap((d) => (d.agents || []).filter((a) => a.role === 'verifier'));
  const firstWave = verAgents.filter((a) => !a.secondWave).length; const secondWave = verAgents.filter((a) => a.secondWave).length;
  const samples = SESSION.filter((x) => x.t >= verStart && x.t <= verEnd).map((x) => ({ ts: new Date(x.t).toISOString(), load1: x.load1 })); const own = fs.readFileSync(path.join(dir, 'load-samples.jsonl'), 'utf8').split('\n').filter(Boolean).map((l) => JSON.parse(l)).filter((x) => Date.parse(x.ts) >= verStart && Date.parse(x.ts) <= verEnd);
  const stage = samples; const ownMean = own.reduce((a, x) => a + x.load1, 0) / own.length; const ownMax = Math.max(...own.map((x) => x.load1));
  const stageMean = stage.reduce((a, s) => a + s.load1, 0) / stage.length;
  const stageMax = Math.max(...stage.map((s) => s.load1));
  const agentMin = ver.reduce((a, s) => a + (s.iend - s.start), 0) / 60000;
  raw[id] = { wallMin, reviewRoundMin, stageMin: (verEnd - verStart) / 60000, ver: ver.length, stopped: stopped.length, cands: cands.length, unverified, firstWave, secondWave, stageMean, stageMax, stageSamples: stage.length, ownMean, ownMax, ownN: own.length, agentMin, startedAt: new Date(Math.min(...starts)).toISOString() };
}

// per-run table rows: wall, review round, verify stage, stage load mean / max
for (const id of IDS) {
  const r = raw[id]; const row = rowsOf(id).join('\n');
  check(`${id} wall (min, from the trace's first start to last end)`, f2(r.wallMin), row);
  check(`${id} review round (min)`, f2(r.reviewRoundMin), row);
  check(`${id} verify stage (min)`, f2(r.stageMin), row);
  check(`${id} verify-stage load1 mean / max (session load.log)`, `${f2(r.stageMean)} / ${f1(r.stageMax)}`, rowsOf(id).find((l) => /\| #\d+ \| (OLD|NEW) \|/.test(l))); out.push(`INFO ${id} own sampler stage load1 mean / max: ${f2(r.ownMean)} / ${f1(r.ownMax)} (${r.ownN} samples; session log ${r.stageSamples})`);
  check(`${id} verifiers dispatched (first + second wave)`, `${r.ver} (${r.firstWave} + ${r.secondWave})`, row);
  check(`${id} verifiers stopped at the deadline`, `${r.stopped}`, row);
  check(`${id} candidates`, `${r.cands} (`, row);
  check(`${id} verifier agent-minutes`, f1(r.agentMin), row);
}
// unverified counts and second-wave tails
const un = (id) => raw[id].unverified;
out.push(`INFO unverified candidates by run: ${IDS.map((i) => `${i.slice(0, 6)}=${un(i)}`).join(' ')}`);
if (un('v26-08-pr314-new-first-redo4') !== 3 || un('v26-09-pr315-new-first') !== 3 || IDS.filter((i) => /old/.test(i)).some((i) => un(i) !== 0) || un('v26-04-pr319-new-first') !== 0) { bad += 1; out.push('FAIL unverified counts differ from the report (3, 3, and none elsewhere)'); } else out.push('OK   unverified: 3 of 5 (run 8), 3 of 7 (run 9), none in the other seven runs');
// second-wave tails: first-wave end to last verifier end
for (const id of ['v26-04-pr319-new-first', 'v26-08-pr314-new-first-redo4', 'v26-09-pr315-new-first']) {
  const spans = readTrace(path.join(RUNS, id)).filter((s) => s.name === 'review.verifier').sort((a, b) => a.start - b.start);
  const dispatch = fs.readFileSync(path.join(RUNS, id, 'dispatch.log.jsonl'), 'utf8').split('\n').filter(Boolean).map((l) => JSON.parse(l));
  const sw = dispatch.flatMap((d) => (d.agents || []).filter((a) => a.role === 'verifier' && a.secondWave).map((a) => a.name));
  const firstEnd = Math.max(...spans.filter((s) => !sw.includes(s.attrs.agent)).map((s) => s.iend)); const lastEnd = Math.max(...spans.map((s) => s.iend)); const start0 = Math.min(...spans.map((s) => s.start));
  const tail = (lastEnd - firstEnd) / 60000; const firstStage = (firstEnd - start0) / 60000;
  check(`${id} first-wave stage`, f2(firstStage)); check(`${id} second-wave tail`, f2(tail));
}
// pairs: wall differences and percentages
const pairs = [['v26-05-pr319-old-first', 'v26-04-pr319-new-first'], ['v26-06-pr314-old-first', 'v26-08-pr314-new-first-redo4'], ['v26-10-pr315-old-first', 'v26-09-pr315-new-first']];
for (const [o, n] of pairs) {
  const dRounded = Number(f2(raw[n].wallMin)) - Number(f2(raw[o].wallMin)); const d = raw[n].wallMin - raw[o].wallMin; const pct = 100 * dRounded / Number(f2(raw[o].wallMin));
  check(`${n} minus ${o}: wall difference (of the two-decimal figures)`, `${dRounded > 0 ? '+' : ''}${f2(dRounded)} min (${pct > 0 ? '+' : ''}${f1(pct)}%)`);
  out.push(`INFO unrounded wall difference ${d.toFixed(3)} min (${(100 * d / raw[o].wallMin).toFixed(2)}%)`); if (Math.abs(d - dRounded) > 0.011) { bad += 1; out.push('FAIL unrounded and displayed differences disagree by more than 0.011 min'); }
}
const sumNew = pairs.reduce((a, [, n]) => a + raw[n].wallMin, 0); const sumOld = pairs.reduce((a, [o]) => a + raw[o].wallMin, 0);
check('sum of the three NEW walls', f2(sumNew)); check('sum of the three OLD walls', f2(sumOld));
check('mean paired difference', `+${f2((sumNew - sumOld) / 3)} min`);
// verifier totals and stops by arm
const sumBy = (arm, k) => IDS.filter((i) => (arm === 'old' ? /-old-/.test(i) : /-new-/.test(i))).reduce((a, i) => a + raw[i][k], 0);
check('stops table, OLD row', `| OLD | 1, 5, 6, 10 | ${sumBy('old', 'ver')} | ${sumBy('old', 'stopped')} (${f1(100 * sumBy('old', 'stopped') / sumBy('old', 'ver'))}%) |`); check('stops table, NEW row', `| NEW (all five runs) | 2, 3, 4, 8, 9 | ${sumBy('new', 'ver')} | ${sumBy('new', 'stopped')} (${f1(100 * sumBy('new', 'stopped') / sumBy('new', 'ver'))}%) |`);
out.push(`INFO stops: OLD ${sumBy('old', 'stopped')} of ${sumBy('old', 'ver')}, NEW ${sumBy('new', 'stopped')} of ${sumBy('new', 'ver')}`);
if (sumBy('old', 'stopped') !== 3 || sumBy('new', 'stopped') !== 8 || sumBy('old', 'ver') !== 59 || sumBy('new', 'ver') !== 49) { bad += 1; out.push('FAIL stop totals differ from the report'); } else out.push('OK   stops: OLD 3 of 59, NEW 8 of 49');
// votes
let real = 0; let notReal = 0; let abst = 0;
for (const id of IDS) for (const c of rd(path.join(RUNS, id, 'summary.json')).ledger.candidates) { real += c.real; notReal += c.notReal; abst += c.abstain; }
check('votes recorded', `${real + notReal + abst} votes, ${real} \`real\`, ${notReal} \`not-real\``);
// harness statuses
let completed = 0; let cancelled = 0;
for (const id of IDS) for (const a of Object.values(rd(path.join(RUNS, id, 'jobs.json')))) { const s = a.final && a.final.status; if (s === 'completed') completed += 1; else cancelled += 1; }
check('harness jobs completed / not completed', `160 agents: ${completed} \`completed\`, ${cancelled} \`cancelled\``);
// evidence trace identical to the repository copy
for (const id of IDS) { const a = fs.readFileSync(path.join(RUNS, id, 'trace.jsonl')); const b = fs.readFileSync(path.join(EVID, id, 'trace.jsonl')); if (!a.equals(b)) { bad += 1; out.push(`FAIL evidence trace differs: ${id}`); } }
out.push('OK   evidence traces are byte-identical to the run directories');
// stored walltime rows
for (const id of IDS) { const row = rd(path.join(EVID, id, 'walltime-row.json')); const r = row.row || row; out.push(`INFO walltime row ${id}: wall_min ${r.wall_min} (trace-derived ${raw[id].wallMin.toFixed(3)})`); if (Math.abs(r.wall_min - raw[id].wallMin) > 0.0015) { bad += 1; out.push(`FAIL walltime row differs for ${id}`); } }
fs.writeFileSync(path.join(HOME, '.omp/pr-review-speedup/v26/final/out/verification.txt'), out.join('\n') + '\n');
console.log(out.join('\n'));
console.log(bad ? `\n${bad} MISMATCH(ES)` : '\nALL CHECKS PASSED');
process.exit(bad ? 1 : 0);
